1. Configuración de Proveedores (app.config.ts)

Registra el adaptador de Cache Storage opcionalmente con un nombre personalizado:

// app.config.ts
import { ApplicationConfig, provideZonelessChangeDetection } from '@angular/core';
import { provideCacheStorage } from 'ngx-storage-manager';

export const appConfig: ApplicationConfig = {
  providers: [
    provideZonelessChangeDetection(),
    provideCacheStorage('mi-cache-personalizada'), // Opcional: define el bucket de caché
  ],
};

2. Inyección y Uso en Componentes / Servicios

Inyecta StorageManagerService y utiliza sus métodos asíncronos tipados:

// mi-componente.ts
import { Component, inject } from '@angular/core';
import { StorageManagerService } from 'ngx-storage-manager';

interface PerfilUsuario {
  id: number;
  nombre: string;
}

@Component({ ... })
export class MiComponente {
  private readonly storage = inject(StorageManagerService);

  async guardar(): Promise<void> {
    // Guardar en LocalStorage
    await this.storage.set('local', 'perfil', { id: 1, nombre: 'Valentina' });

    // Guardar en CacheStorage
    await this.storage.set('cache', 'datos_offline', { version: 2 });
  }

  async leer(): Promise<void> {
    const usuario = await this.storage.get<PerfilUsuario>('local', 'perfil');
    console.log(usuario?.nombre); // 'Valentina'
  }

  async comprobar(): Promise<void> {
    const existe = await this.storage.has('local', 'perfil');
  }

  async eliminar(): Promise<void> {
    await this.storage.remove('local', 'perfil');
  }

  async limpiarTodo(): Promise<void> {
    await this.storage.clear('local'); // Limpia solo las claves 'ngxsm:*'
  }
}
